def VectorSLT.SLT.Z.{u_1} : {α : Type u_1} →
  [inst : MeasurableSpace α] → MeasureTheory.Measure α → (α → ℝ) → ℝ → ENNReal :=
fun {α} [MeasurableSpace α] ν K N => ∫⁻ (x : α), ENNReal.ofReal (Real.exp (-(N * K x))) ∂ν
def VectorSLT.SLT.LaplaceLower : (ℝ → ENNReal) → ℝ → ℕ → Prop :=
fun Zf lam m => ∃ c, 0 < c ∧ ∃ N₀, ∀ (N : ℝ), N₀ ≤ N → ENNReal.ofReal (c * N ^ (-lam) * Real.log N ^ (m - 1)) ≤ Zf N
def VectorSLT.SLT.LaplaceUpper : (ℝ → ENNReal) → ℝ → ℕ → Prop :=
fun Zf lam m => ∃ C N₀, ∀ (N : ℝ), N₀ ≤ N → Zf N ≤ ENNReal.ofReal (C * N ^ (-lam) * Real.log N ^ (m - 1))
def VectorSLT.SLT.HasLaplacePair : (ℝ → ENNReal) → ℝ → ℕ → Prop :=
fun Zf lam m => VectorSLT.SLT.LaplaceLower Zf lam m ∧ VectorSLT.SLT.LaplaceUpper Zf lam m
def VectorSLT.layer : {n : ℕ} → Matrix (Fin n) (Fin n) ℝ → (Fin n → ℝ) → Fin n → ℝ :=
fun {n} B z => z + B.mulVec z ^ 3
def VectorSLT.net : VectorSLT.Raw16 → (Fin 2 → ℝ) → Fin 2 → ℝ :=
fun θ => VectorSLT.prefixNet (VectorSLT.weights θ) 4
def VectorSLT.popLossA : MeasureTheory.Measure (Fin 2 → ℝ) → Matrix (Fin 2) (Fin 2) ℝ → VectorSLT.Raw16 → ℝ :=
fun μ A θ => ∫ (x : Fin 2 → ℝ), ∑ k, (VectorSLT.net θ x k - VectorSLT.layer A x k) ^ 2 ∂μ
def VectorSLT.Zpos : MeasureTheory.Measure (Fin 2 → ℝ) →
  Matrix (Fin 2) (Fin 2) ℝ → (VectorSLT.Raw16 → ENNReal) → (VectorSLT.Raw16 → ENNReal) → Fin 4 → ℝ → ENNReal :=
fun μ A φ χ p N =>
  ∫⁻ (θ : VectorSLT.Raw16),
    ENNReal.ofReal (Real.exp (-(N * VectorSLT.popLossA μ A θ))) * (φ θ * χ (θ - VectorSLT.basePos A p))
def VectorSLT.basePos : Matrix (Fin 2) (Fin 2) ℝ → Fin 4 → VectorSLT.Raw16 :=
fun A p =>
  VectorSLT.ofEnt fun i j =>
    (if p = 0 then A i j else 0, if p = 1 then A i j else 0, if p = 2 then A i j else 0, if p = 3 then A i j else 0)
def VectorSLT.ShearTarget : Matrix (Fin 2) (Fin 2) ℝ → Prop :=
fun A => ∃ a v, (∀ (i j : Fin 2), A i j = a i * v j) ∧ a 0 ≠ 0 ∧ a 1 ≠ 0 ∧ v ≠ 0 ∧ v 0 * a 0 ^ 3 + v 1 * a 1 ^ 3 = 0
def VectorSLT.ThmK : MeasureTheory.Measure (Fin 2 → ℝ) →
  Matrix (Fin 2) (Fin 2) ℝ → (VectorSLT.Raw16 → ENNReal) → (VectorSLT.Raw16 → ENNReal) → Prop :=
fun μ A φ χ =>
  ∃ c C N₀,
    0 < c ∧
      ∀ (N : ℝ),
        N₀ ≤ N →
          ENNReal.ofReal (c * N⁻¹ ^ 4) ≤ VectorSLT.Zpos μ A φ χ 3 N ∧
            VectorSLT.Zpos μ A φ χ 3 N ≤ ENNReal.ofReal (C * N⁻¹ ^ 4)
def VectorSLT.ThmKUpper : MeasureTheory.Measure (Fin 2 → ℝ) →
  Matrix (Fin 2) (Fin 2) ℝ → (VectorSLT.Raw16 → ENNReal) → (VectorSLT.Raw16 → ENNReal) → Prop :=
fun μ A φ χ => ∃ C N₀, ∀ (N : ℝ), N₀ ≤ N → VectorSLT.Zpos μ A φ χ 3 N ≤ ENNReal.ofReal (C * N⁻¹ ^ 4)
def VectorSLT.ThmN : MeasureTheory.Measure (Fin 2 → ℝ) →

def VectorSLT.Meta : ℝ → ℝ → ℝ :=
fun η N => if η = 0 then N else min N |η|⁻¹
def VectorSLT.Leta : ℝ → ℝ → ℝ :=
fun η N => Real.log (VectorSLT.Meta η N)
